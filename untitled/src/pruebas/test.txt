PROGRAM PruebaTodo
VAR
  s1, s2: STRING;
  x, y, z: INT;
BEGIN
  s1 := "hola";
  s2 := s1;

  x := 5;
  y := 3;

  z := x + y;      
  z := z - 2;      

  x := x++;     
  y := y--;      

  IF z = x THEN
    BEGIN
      s1 := "iguales";
    END
  ELSE
    BEGIN
      s1 := "distintos";
    END
END
